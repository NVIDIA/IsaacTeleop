# Python bindings for Televiz — _viz.so + isaacteleop.viz package.
# Output lands in ${CMAKE_BINARY_DIR}/python_package/<cfg>/isaacteleop/viz/
# next to the __init__.py staged by src/python/CMakeLists.txt.

cmake_minimum_required(VERSION 3.20)

# Bindings are split per-sub-module to mirror the C++ dep DAG
# (core → layers → session). viz_bindings.cpp is the PYBIND11_MODULE
# entry point and calls into the bind_*(m) helpers in the others.
pybind11_add_module(viz_py
    viz_bindings.cpp
    core_bindings.cpp
    layers_bindings.cpp
    session_bindings.cpp
)

target_link_libraries(viz_py
    PRIVATE
        viz::session
        viz::layers
        viz::core
)

set_target_properties(viz_py PROPERTIES
    OUTPUT_NAME "_viz"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python_package/$<CONFIG>/isaacteleop/viz"
)

# The .pyi stub for isaacteleop.viz._viz is generated by the central stubgen
# command in src/core/python/CMakeLists.txt, alongside every other module's.
# It must not be a POST_BUILD hook here: importing _viz runs the parent
# isaacteleop/__init__.py, which eagerly imports every sibling extension
# module, and POST_BUILD carries no dependency edges — it fired the moment
# viz_py linked and raced the siblings' links under `-j`.
